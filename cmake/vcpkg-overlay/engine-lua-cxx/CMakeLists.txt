cmake_minimum_required(VERSION 3.28)
project(engine_lua_cxx VERSION 5.4.9 LANGUAGES CXX)
include(GNUInstallDirs)
file(GLOB lua_sources CONFIGURE_DEPENDS src/*.c)
list(REMOVE_ITEM lua_sources "${CMAKE_CURRENT_SOURCE_DIR}/src/lua.c" "${CMAKE_CURRENT_SOURCE_DIR}/src/luac.c")
set_source_files_properties(${lua_sources} PROPERTIES LANGUAGE CXX)
add_library(engine_lua_cxx STATIC ${lua_sources})
set_target_properties(engine_lua_cxx PROPERTIES OUTPUT_NAME lua-cxx EXPORT_NAME lua)
target_compile_features(engine_lua_cxx PRIVATE cxx_std_20)
target_include_directories(engine_lua_cxx PUBLIC
    "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src>"
    "$<INSTALL_INTERFACE:include/engine-lua-cxx>"
)
if(WIN32)
    target_compile_definitions(engine_lua_cxx PRIVATE _CRT_SECURE_NO_WARNINGS)
elseif(UNIX)
    target_compile_definitions(engine_lua_cxx PRIVATE LUA_USE_POSIX LUA_USE_DLOPEN)
    target_link_libraries(engine_lua_cxx PRIVATE ${CMAKE_DL_LIBS} m)
endif()
install(TARGETS engine_lua_cxx EXPORT engine_lua_cxxTargets ARCHIVE DESTINATION lib)
install(FILES src/lua.h src/luaconf.h src/lauxlib.h src/lualib.h DESTINATION include/engine-lua-cxx)
install(EXPORT engine_lua_cxxTargets NAMESPACE engine_lua_cxx:: DESTINATION share/engine_lua_cxx)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/engine_lua_cxxConfig.cmake"
    [=[include("${CMAKE_CURRENT_LIST_DIR}/engine_lua_cxxTargets.cmake")]=])
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/engine_lua_cxxConfig.cmake" DESTINATION share/engine_lua_cxx)
