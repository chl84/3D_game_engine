Reproducer fault.c:
#define _GNU_SOURCE
#include <dlfcn.h>
#include <errno.h>
#include <stdio.h>
#include <string.h>
#include <unistd.h>
int fsync(int fd) {
    char descriptor[64], path[1024];
    snprintf(descriptor, sizeof(descriptor), "/proc/self/fd/%d", fd);
    ssize_t n = readlink(descriptor, path, sizeof(path)-1);
    static int journal_syncs;
    if (n > 0) {
        path[n] = 0;
        if (strstr(path, "live-commands-") && n >= 6 && strcmp(path+n-6, ".jsonl") == 0 && ++journal_syncs == 2) {
            errno = EIO;
            return -1;
        }
    }
    int (*real_fsync)(int) = (int (*)(int))dlsym(RTLD_NEXT, "fsync");
    return real_fsync(fd);
}

Reproducer fault-check.py (temporary paths are local diagnostic artifacts):
import json, os, pathlib, socket, subprocess, tempfile

with tempfile.TemporaryDirectory(prefix='journal-sync-fault-') as t:
    project=pathlib.Path(t)/'project'
    subprocess.run([engine,'project','create',str(project),'--json'],check=True,capture_output=True)
    p=subprocess.Popen([engine,'live','serve',str(project),'--json'],stdout=subprocess.PIPE,stderr=subprocess.PIPE,text=True,env=dict(os.environ,LD_PRELOAD='/tmp/m37-journal-fsync-fault.so'))
    try:
        ready=json.loads(p.stdout.readline())
